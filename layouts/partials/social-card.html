{{/* Builds an on-brand "social card" OG/Twitter preview image for any page: the page's own photo (seo.featured_image, else featuredImg, else thumbImg, else the site default) composited with a dark gradient scrim, the FTSK crest and the page title - so a shared link looks intentional, not just a raw photo. Returns an absolute image URL (falls back to the plain site default image on any missing/unprocessable source, so a bad photo never breaks the build). Requires static/images/* to also be mounted at assets/images/* (see the extra [[module.mounts]] entry in config.toml) so these already-published static photos can run through Hugo's image processing pipeline too. */}} {{ $page := . }} {{ $fallback := partial "absurl.html" hugo.Data.meta.image }} {{ $result := $fallback }} {{ $srcPath := "" }} {{ with $page.Params.seo.featured_image }}{{ $srcPath = . }}{{ end }} {{ if not $srcPath }}{{ with $page.Params.featuredImg.image_path }}{{ $srcPath = . }}{{ end }}{{ end }} {{ if not $srcPath }}{{ with $page.Params.thumbImg.image_path }}{{ $srcPath = . }}{{ end }}{{ end }} {{ if not $srcPath }}{{ $srcPath = hugo.Data.meta.image }}{{ end }} {{ with resources.Get ($srcPath | strings.TrimPrefix "/") }} {{ $img := . }} {{ with resources.Get "images/og/gradient.png" }} {{ $gradient := . }} {{ with resources.Get "images/ftsk-logo.png" }} {{ $logo := .Resize "0x100" }} {{ $titleSize := 54 }} {{ if gt (len $page.Title) 22 }}{{ $titleSize = 42 }}{{ end }} {{ if gt (len $page.Title) 34 }}{{ $titleSize = 34 }}{{ end }} {{ $logoX := 60 }} {{ $logoY := 490 }} {{ $textX := 150 }} {{ $gradientFilter := images.Overlay $gradient 0 0 }} {{ $logoFilter := images.Overlay $logo $logoX $logoY }} {{ $titleFilter := images.Text $page.Title (dict "x" $textX "y" 470 "size" $titleSize "color" "f2ede1" "aligny" "bottom") }} {{ $siteFilter := images.Text "FTSK Barlangkutató Szakosztály" (dict "x" $textX "y" (add $logoY 50) "size" 22 "color" "bea070" "aligny" "center") }} {{ $composited := $img.Fill "1200x630 center" | images.Filter $gradientFilter | images.Filter $logoFilter | images.Filter $titleFilter | images.Filter $siteFilter }} {{ $result = $composited.Permalink }} {{ end }} {{ end }} {{ end }} {{ return $result }}
