{{/* Turns a slide's "start zoom" + "animation type/direction/amount" (see data/hero_images.yaml) into an explicit from->to pair for every transform channel (zoom, pan-x, pan-y, rotate) - hero.hugo.html calls this once per breakpoint tier (base/tablet/mobile) so each tier can have a completely different animation, not just tweaked numbers. Returns a dict: zoomFrom/zoomTo, panXFrom/panXTo, panYFrom/panYTo, rotFrom/rotTo. */}} {{ $zoom := .zoom }} {{ $type := .type }} {{ $direction := .direction }} {{ $amount := .amount }} {{ $focus := .focus | default "50% 50%" }} {{ $mirrorX := .mirrorX | default 1.0 }} {{ $mirrorY := .mirrorY | default 1.0 }} {{ $zoomFrom := $zoom }} {{ $zoomTo := $zoom }} {{ $panXFrom := 0.0 }} {{ $panXTo := 0.0 }} {{ $panYFrom := 0.0 }} {{ $panYTo := 0.0 }} {{ $rotFrom := 0.0 }} {{ $rotTo := 0.0 }} {{ if eq $type "pan" }} {{/* _hero.scss pivots the pan/zoom transform around the SAME point as the focus (transform-origin), not the box center - so panning translates the whole already-zoomed slide around that off-center point. If the focus sits close to the edge being panned toward, the translate can push that edge inside the frame, exposing the background behind it (a "crop" gap). Clamp `amount` down to the largest value that's still guaranteed to fully cover the frame, given the start zoom, mirror sign, and how close the focus is to that edge. */}} {{ $isHorizontal := or (eq $direction "left") (eq $direction "right") }} {{ $focusParts := split $focus " " }} {{ $fx := 50.0 }} {{ $fy := 50.0 }} {{ if ge (len $focusParts) 2 }} {{ $fx = replaceRE "%" "" (index $focusParts 0) | float }} {{ $fy = replaceRE "%" "" (index $focusParts 1) | float }} {{ end }} {{ $focusPct := cond $isHorizontal $fx $fy }} {{ $mirrorSign := cond $isHorizontal $mirrorX $mirrorY }} {{ $p := div $focusPct 100.0 }} {{ $q := math.Max $zoom 1.0 }} {{ $safeAmount := 0.0 }} {{ if lt $mirrorSign 0.0 }} {{ $tMin := sub (add (div $p $q) $p) 1.0 }} {{ $tMax := sub (add (div $p $q) $p) (div 1.0 $q) }} {{ $safeAmount = mul (math.Max 0.0 (math.Min (mul $tMin -1.0) $tMax)) 100.0 }} {{ else }} {{ $safeAmount = mul (div (mul (math.Min $p (sub 1.0 $p)) (sub $q 1.0)) $q) 100.0 }} {{ end }} {{ $amount = math.Min $amount $safeAmount }} {{ if eq $direction "left" }} {{ $panXFrom = $amount }} {{ $panXTo = mul $amount -1.0 }} {{ else if eq $direction "up" }} {{ $panYFrom = $amount }} {{ $panYTo = mul $amount -1.0 }} {{ else if eq $direction "down" }} {{ $panYFrom = mul $amount -1.0 }} {{ $panYTo = $amount }} {{ else }} {{/* "right" is the default direction */}} {{ $panXFrom = mul $amount -1.0 }} {{ $panXTo = $amount }} {{ end }} {{ else if eq $type "zoom-in" }} {{ $zoomTo = add $zoom $amount }} {{ else if eq $type "zoom-out" }} {{ $zoomTo = sub $zoom $amount }} {{ if lt $zoomTo 1.0 }}{{ $zoomTo = 1.0 }}{{ end }} {{ else if eq $type "tilt" }} {{ if eq $direction "left" }} {{ $rotFrom = $amount }} {{ $rotTo = mul $amount -1.0 }} {{ else }} {{/* "right" is the default direction */}} {{ $rotFrom = mul $amount -1.0 }} {{ $rotTo = $amount }} {{ end }} {{ end }} {{/* type "none" (or anything unrecognised) just keeps the defaults above: a fully static frame at the defined start zoom, no pan/rotate. */}} {{ return dict "zoomFrom" $zoomFrom "zoomTo" $zoomTo "panXFrom" $panXFrom "panXTo" $panXTo "panYFrom" $panYFrom "panYTo" $panYTo "rotFrom" $rotFrom "rotTo" $rotTo }}
